# -----------------------------------------------------------------------------
# 01. MANIPULACIÓN Y PROCESAMIENTO DE DATOS
# -----------------------------------------------------------------------------
# pandas:  tablas, uniones entre catálogos y agregaciones.
# numpy:   cálculo numérico.
# pyarrow: caché Parquet de Dataset.csv (lectura ~20 veces más rápida).
pandas==3.0.5
numpy==2.5.3
pyarrow==26.0.0


# -----------------------------------------------------------------------------
# 02. MACHINE LEARNING
# -----------------------------------------------------------------------------
# scikit-learn: modelos, validación agrupada, métricas y pipelines.
# lightgbm:     gradient boosting (solo CPU con el wheel de pip).
# joblib:       serialización de los modelos entrenados.
scikit-learn==1.9.0
lightgbm==4.7.0
joblib==1.5.3


# -----------------------------------------------------------------------------
# 03. MACHINE LEARNING — GPU (BOOSTING)
# -----------------------------------------------------------------------------
# Requisito: controlador NVIDIA instalado (nvidia-smi en el PATH). Sin GPU,
# ambos modelos se entrenan en CPU con el mismo código.
#
# xgboost:
#   Gradient Boosting en GPU (device="cuda:i"). Se fija la 2.1.x: el wheel de
#   la 3.x solo trae binarios de GPU para tarjetas Volta o más nuevas (compute
#   capability >= 7.0) y deja fuera las Pascal (GTX 10xx). La 2.1.4 soporta GPU
#   desde compute 6.0. Una GPU por modelo: las particiones de la validación
#   cruzada se reparten entre todas las GPU y se entrenan en paralelo.
#
# catboost:
#   Gradient Boosting. El wheel de pip trae soporte de GPU y entrena cada
#   modelo con todas las GPU a la vez (devices="0:1").
xgboost==2.1.4
catboost==1.2.10


# -----------------------------------------------------------------------------
# 04. ESTADÍSTICA, ASOCIACIÓN Y REDES
# -----------------------------------------------------------------------------
# scipy:       pruebas de Fisher y Mann-Whitney.
# statsmodels: corrección por comparaciones múltiples (Benjamini-Hochberg).
# mlxtend:     FP-growth y reglas de asociación.
# networkx:    redes de coocurrencia (exportación GraphML).
# umap-learn:  proyección no lineal para visualización.
scipy==1.18.1
statsmodels==0.15.0
mlxtend==0.25.0
networkx==3.7
umap-learn==0.5.12
