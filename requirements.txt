# -----------------------------------------------------------------------------
# 01. MANIPULACIÓN Y PROCESAMIENTO DE DATOS
# -----------------------------------------------------------------------------
#
# pandas:
#   Manipulación, transformación y análisis de datos tabulares.
#
# numpy:
#   Computación numérica y estructuras multidimensionales.
#
pandas==3.0.5
numpy==2.5.2


# -----------------------------------------------------------------------------
# 02. MACHINE LEARNING — CORE
# -----------------------------------------------------------------------------
#
# scikit-learn:
#   Algoritmos de Machine Learning, preprocesamiento, selección de modelos,
#   validación cruzada y métricas de evaluación.
#
# joblib:
#   Serialización de modelos/objetos y ejecución paralela utilizada por
#   diferentes componentes del ecosistema científico de Python.
#
scikit-learn==1.9.0
joblib==1.5.3


# -----------------------------------------------------------------------------
# 03. MACHINE LEARNING — BOOSTING Y ENSEMBLES
# -----------------------------------------------------------------------------
#
# xgboost:
#   Gradient Boosting optimizado para problemas de clasificación y regresión.
#   Se fija la 2.1.x: el wheel de la 3.x solo trae binarios de GPU para tarjetas
#   Volta o más nuevas (compute capability >= 7.0) y deja fuera las Pascal
#   (GTX 10xx). La 2.1.4 soporta GPU desde compute 6.0.
#
# lightgbm:
#   Gradient Boosting basado en árboles, optimizado para rendimiento y
#   conjuntos de datos de gran tamaño. El wheel de pip es solo CPU (la GPU
#   requiere compilar con -DUSE_GPU=1).
#
# catboost:
#   Gradient Boosting basado en árboles, con soporte especialmente robusto
#   para variables categóricas. El wheel de pip ya trae soporte de GPU.
#
xgboost==2.1.4
lightgbm==4.7.0
catboost==1.2.10


# -----------------------------------------------------------------------------
# 04. MACHINE LEARNING — DEEP LEARNING
# -----------------------------------------------------------------------------
#
# torch:
#   Framework de Deep Learning para construcción y entrenamiento de redes
#   neuronales. Se fija la rueda CUDA 12.6 para aprovechar las GPU NVIDIA
#   compatibles; la rueda CPU no permite detectar ni utilizar CUDA.
#
--extra-index-url https://download.pytorch.org/whl/cu126
torch==2.14.0+cu126


# -----------------------------------------------------------------------------
# 05. MACHINE LEARNING — APRENDIZAJE NO SUPERVISADO
